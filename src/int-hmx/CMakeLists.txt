include(${HEXAGON_SDK_ROOT}/build/cmake/hexagon_fun.cmake)
if(CMAKE_SYSTEM_NAME STREQUAL "Android")
    add_library(int_hmx_stub OBJECT ${CMAKE_CURRENT_BINARY_DIR}/int_hmx_iface_stub.c)
    target_include_directories(int_hmx_stub PUBLIC ${HEXAGON_SDK_ROOT}/incs ${HEXAGON_SDK_ROOT}/incs/stddef ${CMAKE_CURRENT_BINARY_DIR})
    build_idl(int_hmx_iface.idl int_hmx_stub)
    add_executable(int_hmx_micro_probe ../../tools/int_hmx_micro_probe.cpp)
    target_compile_features(int_hmx_micro_probe PRIVATE cxx_std_17)
    target_compile_options(int_hmx_micro_probe PRIVATE -Wall -Wextra -Werror)
    target_include_directories(int_hmx_micro_probe PRIVATE ${HEXAGON_SDK_ROOT}/incs ${HEXAGON_SDK_ROOT}/incs/stddef ${CMAKE_CURRENT_BINARY_DIR} /baseline/vendor /baseline/ggml/include)
    target_link_libraries(int_hmx_micro_probe PRIVATE int_hmx_stub ${HEXAGON_SDK_ROOT}/ipc/fastrpc/remote/ship/android_aarch64/libcdsprpc.so /baseline/pkg-adb/llama.cpp/lib/libggml.so /baseline/pkg-adb/llama.cpp/lib/libggml-base.so log dl)
else()
    add_library(int_hmx_htp SHARED int_hmx_service.c ${CMAKE_CURRENT_BINARY_DIR}/int_hmx_iface_skel.c)
    target_include_directories(int_hmx_htp PRIVATE ${HEXAGON_SDK_ROOT}/incs ${HEXAGON_SDK_ROOT}/incs/stddef ${CMAKE_CURRENT_BINARY_DIR})
    target_include_directories(int_hmx_htp SYSTEM PRIVATE /hexkl/include)
    target_compile_options(int_hmx_htp PRIVATE -O3 -mhvx -mhvx-length=128B -Wall -Wextra -Werror)
    target_link_libraries(int_hmx_htp PRIVATE /hexkl/lib/hexagon_toolv19_v75/libhexkl_micro.a)
    build_idl(int_hmx_iface.idl int_hmx_htp)
endif()
